from fastapi import FastAPI from fastapi.staticfiles import StaticFiles from fastapi.responses import FileResponse from google import genai import os app = FastAPI() # static 폴더 안의 파일들을 웹에서 접근할 수 있게 열어줍니다. app.mount("/static", StaticFiles(directory="static"), name="static") # 제미나이 클라이언트 초기화 (환경 변수 또는 시스템에 설정된 키를 자동으로 사용합니다) client = genai.Client() @app.get("/") def read_index(): # 주소창에 딱 들어왔을 때 아들이 볼 수 있는 첫 화면(index.html)을 보여줍니다. return FileResponse("static/index.html") @app.get("/generate-problem") def generate_problem(): try: # 제미나이에게 중학교 1학년 수준의 수학 문제를 요청합니다. response = client.models.generate_content( model='gemini-2.5-flash', contents='중학교 1학년 수준의 수학 연산 또는 방정식 문제 1개를 만들어줘. 반드시 딱 문제 내용과 정답만 알아보기 쉽게 간결하게 줘.', ) ai_text = response.text # 임시로 문제와 정답을 전달하는 구조 (필요에 따라 파싱 로직을 다듬을 수 있습니다) return { "problem": ai_text, "answer": "확인" } except Exception as e: return {"problem": f"문제 생성 중 오류가 발생했습니다: {str(e)}", "answer": ""}
